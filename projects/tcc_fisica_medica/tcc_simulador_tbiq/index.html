🚀 Criador de Novo Simulador (Template de Código)

O TBIQ Studio possui arquitetura em plugins. Qualquer aluno pode criar um novo simulador herdando de SimulatorBase:

// 1. Crie seu arquivo: js/modules/MeuNovoSimulador.js
import { SimulatorBase } from '../core/SimulatorBase.js';
import { globalRegistry } from '../core/SimulatorsRegistry.js';

export class MeuNovoSimulador extends SimulatorBase {
    constructor() {
        super({
            id: 'meu_novo_simulador',
            title: 'Análise de Artefatos em TC',
            badge: 'Módulo Customizado',
            clinicalScenario: 'Detecção de Artefatos Metálicos',
            scenarioDesc: 'Avaliação de próteses de titânio e redução de endurecimento de feixe.'
        });
    }

    getParameters() {
        return {
            proteseMaterial: { type: 'select', options: ['Titânio', 'Aço Inox', 'Zircônia'], default: 'Titânio', label: 'Material' },
            marLevel: { type: 'range', min: 0, max: 100, default: 50, label: 'Algoritmo MAR (%)' }
        };
    }

    synthesizeSlice(w, h, params, math) {
        const slice = new Float32Array(w * h);
        // ... sua física tomográfica em HU ...
        return slice;
    }

    renderCharts(c1, c2, params, math) {
        // ... seus gráficos analíticos ...
    }
}

// 2. Registre em 1 linha:
globalRegistry.register(new MeuNovoSimulador());
